usage: unemu --version | --about (licence, source and third-party notices)
       unemu run --machine FILE [--store DIR | --no-store] [--rom-dir DIR]... [--max-steps N] [--idle-limit N] [--diag-at N] [--trace-exceptions] [--print CARD] [--allow-unlisted-firmware]
         --diag-at N writes the machine's diagnostic report to stderr when the run reaches step N (the unit of --max-steps), then runs on
         --trace-exceptions writes the exceptions and interrupts the guest takes to stderr when the run ends: the last 10000 entries, each with the registers of the interrupted code (the bytes at CS:IP as at the end of each run slice), consecutive identical deliveries folded into one with a count and the last one's registers, and a summary per vector
       unemu image import FILE --store DIR --tag NAME:TAG [--create] [--no-compress] [--rom-dir DIR]... [--recover MEDIUM=tip|MEDIUM=generation:N]... [--min-app-version X.Y.Z]
         --min-app-version (here, on image media import and on image firmware import and edit) raises the oldest unemupc the image declares it needs: never below what this release computes, never above this release
       unemu image compress NAME:TAG --store DIR --tag NAME:TAG
       unemu image ls --store DIR
       unemu image sweep --store DIR
       unemu image run NAME:TAG --store DIR --workspace NAME [--new] [--max-steps N]
       unemu image workspace ls --store DIR
       unemu image workspace discard NAME --store DIR --yes
       unemu image workspace abandon-publication NAME --store DIR
       unemu image publish --from-workspace NAME --store DIR --tag NAME:TAG [--flatten]
       unemu image machine ls --store DIR
       unemu image machine export ID [--store DIR]
       unemu image machine export ID --slot NAME --to FILE [--store DIR]
       unemu image machine new NAME:TAG --store DIR [--name NAME]
       unemu image machine rename ID NAME --store DIR
       unemu image machine duplicate ID --store DIR [--name NAME]
       unemu image machine reset ID --store DIR --yes
       unemu image machine rm ID --store DIR --yes
       unemu image machine settings ID --store DIR [--set PATH=VALUE]... [--clear] [--dry-run]
       unemu image machine commit ID --store DIR --tag NAME:TAG [--title TITLE] [--keep-on-old-image]
       unemu image machine run ID --store DIR [--max-steps N]
       unemu image media import FILE...|DIR --title TITLE --tag NAME:TAG --store DIR [--create] [--dry-run] [--no-compress] [--label N=TEXT]... [--format N=FORMAT]... [--extra PATH[:DOSNAME]]... [--open --license LicenseRef-NAME --license-text EXTRA] [--open --license EXPRESSION --license-text EXTRA --source EXTRA] [--min-app-version X.Y.Z] [SET-METADATA]
         SET-METADATA: [--publisher P]... [--developer D]... [--year YYYY] [--genre G]... [--platform PLATFORM] [--description TEXT] [--version V] [--edition E] [--set-label L]...
                       [--need KIND=VALUE[,VALUE]...]... [--requirements TEXT] [--work ID] [--synopsis TEXT] [--work-id wikidata|mobygames|igdb=ID]...
       unemu image firmware import FOLDER|FILES... [--store DIR] [--create] [--rescan] [--dry-run] [--for CATALOGUE-ID [EXTRAS] [METADATA] [--min-app-version X.Y.Z]]
       unemu image firmware import FILE --unlisted ID --role ROLE [--store DIR] [--create] [--dry-run] [EXTRAS] [METADATA] [--min-app-version X.Y.Z]
       unemu image firmware edit NAME:TAG [--to NAME:TAG] [--store DIR] [--regenerate-readme] [METADATA] [--min-app-version X.Y.Z]
         EXTRAS: [--extra PATH[:DOSNAME] [--extra-license SPDX]]...
         METADATA: [--title T] [--description T] [--notes T] [--vendor T] [--version T] [--date YYYY-MM-DD]
                   [--label L]... [--board B]... [--class C]... [--requires KIND]...
                   [--dumped-from T | --downloaded-from T | --built-from T] [--revision R] [--url U] [--provenance-date YYYY-MM-DD]
                   [--open [--license SPDX] | --user-supplied] [--source ARCHIVE.tar|.zip [--source-license SPDX]]
       unemu image firmware ls [--store DIR]
       unemu image firmware show NAME:TAG [--store DIR]
       unemu image media ls --store DIR
       unemu image media show NAME:TAG --store DIR
       unemu image export NAME:TAG --to DIR [--files] [--store DIR] (every layer as a plain file under its file name, with config.toml, manifest.json, EXPORT-VERSION and SHA256SUMS; --files also the files of the set's FAT12 disks; exit 1 refused, nothing overwritten)
       unemu image push NAME:TAG [REFERENCE] --store DIR [--registry HOST] [--move] [--dry-run] [--config FILE]
       unemu image pull REFERENCE --store DIR [--tag NAME:TAG] [--create] [--registry HOST] [--config FILE]
       unemu image remote ls REPOSITORY|REGISTRY [--registry HOST] [--config FILE]
       unemu image login HOST --user NAME [--config FILE]
       unemu image registry set HOST [--private | --namespace NAME[:private|:public]...] [--config FILE]
       unemu image template build --store DIR [--vendor DIR]... [--dry-run]
       unemu recipe check FILE|FOLDER... [--deny-drift] (the static checks of recipes and work files; exit 0 clean, 1 with findings)
       unemu recipe show FILE (a media recipe's title record merged with its work file; exit 1 with the check's findings)
       unemu recipe plan FILE|REF [--json] [--library DIR]... [--store DIR] [--config FILE] (where each input is found, or why not; a REF is judged from its manifest and config; exit 0 all found, 1 missing)
       unemu recipe build FILE [--fetch] [--library DIR]... [--store DIR] [--config FILE] [--keep-as TAG] [--deny-drift] (builds a recipe and tags it, a pinned one from its lock; --fetch downloads missing inputs; exit 0 tagged, 1 not built)
       unemu recipe pin FILE [--fetch] [--library DIR]... [--store DIR] [--config FILE] (builds a draft, writes its lock beside it and its digest into it; exit 0 pinned, 1 not built)
       unemu recipe conformance FILE|FOLDER... [--library DIR]... [--store DIR] [--config FILE] [--allow-drafts] (builds each pinned recipe into a scratch store and compares its digest; exit 0 all conform, 1 not)
       unemu recipe scaffold FILE|FOLDER... (the [input] tables, member lists with the archive's times and file sets of files the author has; exit 1 for one a recipe cannot hold)
       unemu recipe inputs FILE --format dat|csv|sha256sums (a recipe's inputs for a ROM manager or sha256sum -c; exit 1 with the check's findings)
       unemu recipe push FILE REF [--dry-run] [--registry HOST] [--store DIR] [--config FILE] (publishes a pinned recipe with its work file, bundled inputs and lock under the recipe guard; exit 1 refused)
       unemu recipe pull REF [--to DIR] [--registry HOST] [--config FILE] (writes a recipe and its parts into DIR, where it plans and builds; exit 1 refused)
       unemu library ls [--store DIR] [--config FILE]
       unemu library add|remove DIR [--config FILE]
       unemu library scan [DIR...] [--store DIR] [--config FILE] (lists files, zip directories and catalogues into the index; hashes nothing)
       unemu media create floppy --type 1.44M|1.2M|720K|360K OUT [--format] [--label NAME] [--force]
       unemu media create floppy --type 1.44M|1.2M|720K|360K OUT --add PATH [--add PATH]... [--top NAME] [--label NAME] [--split] [--force]
         --add copies host files and folders (folders whole, symlinks never followed) onto a new FAT12 disk with 8.3 names;
         --top NAME puts them all under one folder; --split writes <stem>-1ofN.<ext>, <stem>-2ofN.<ext>... when they need more than one disk
       unemu media create hd (--chs C/H/S | --size SIZE[MiB|GiB]) OUT [--force]
       unemu media generations --base IMG --journal J
       unemu media verify --base IMG --journal J [--repair-tail --backup PATH]
       unemu media rollback --base IMG --journal J --to N --backup PATH
       unemu media commit (alias compact) --base IMG --journal J --to N --output OUT
